  myrel A = record { dis : level->A->Prop;  rel : level->A->A->Prop;
                     + 3 laws: equiv(rel l) ; order-monotone rel ; order-monotone
                     dis ; and dis l a0 -> (dis l a1 <-> rel l a0 a1) }.
  Build with refine (@MyRel _ dis rel _ _ _ _).  ssreflect: fst=.1, snd=.2.

  rel_refl/sym/trans, dis_rel_dis (dis x -> rel x y -> dis y),
  dis_rel_dis2 (dis x -> rel y x -> dis y),
  myrel_rule1 (order l0 l1 -> rel l1 -> rel l0),
  myrel_rule2 (order l0 l1 -> dis l1 -> dis l0).

  eqpair_R I O IRel ORel : myrel [Times I O]  -- RIGHT-keyed disclosure:
     rel = (rel IRel x.1 y.1 /\ rel ORel x.2 y.2) \/ (dis ORel x.2 /\ dis ORel y.2)
     dis = dis ORel x.2
  eqpair_LR : LEFT-keyed (dis ARel x.1).  rel_eqpair_LR / dis_eqpair_LR split them.
  eqpair I O : the "both must be related" pair (no disclosed branch) -- sta output.

  NI IRel ORel p = forall l, NI_l ... ; NI_l has 3 clauses:
    (1 rel)    rel IRel l i i' -> Trace(insert n (inl i) t) -> Trace(insert n (inl i') t)
    (2 insert) dis IRel l i    -> Trace t                   -> Trace(insert n (inl i) t)
    (3 remove) dis IRel l i    -> Trace(insert n (inl i) t) -> Trace t
  NI_reduceI / NI_reduceO peel a reduceI/reduceO step off an NI.

  insert n a l:  n=0 -> a::l ; (S n') on [] -> [] ; (S n') on a'::l' -> a'::insert n' a l'.
    *** insert DROPS a when n > size l (out-of-range = no-op). This is the source
        of the clause-2/3 "wrinkle" below. ***
  lthread VRel f g l v L: per-element rel chain; for inl wi at the head it gives
    rel VRel l wi.1 (f wi.2 v) and recurses with state (f wi.2 v).
  projIl/projOl: map a seq of sums; size-preserving.
  sta_proj_lthread: Trace(eqpair VRel ORel) of (sta f g v p) <-> exists T,
     ts=projIl T /\ Trace ORel (projOl T) p /\ lthread ... T.
  projIl_insert_inv: projIl T = insert n (inl i) t -> exists w T'',
     T = insert n (inl (w,i)) T'' /\ projIl T''=t.
  lthread_insert_dis / lthread_remove_dis / lthread_swap / lthread_stable: as named.

  swi I O : bool -> Proc I (Times Bool O) -> Proc (Times Bool I) (Option O).
  swi_trace: Trace(eqmaybe_swi ORel BRel) (swi b p) -> exists t',
     Trace (eqpair_LR BRel ORel) l t' p /\ SwiTrace ORel BRel l b t t'.   <-- eqpair_LR!
  swi_conv: aware BRel true l -> Trace (eqpair_LR BRel ORel) l t_p p ->
     SwiTrace ... -> Trace(eqmaybe_swi ORel BRel) l t (swi b p).           <-- eqpair_LR!
  aware BRel v l = forall v', rel BRel l v v' -> v=v' /\ ~ dis BRel l v'.
  oblivious_NI_l, oblivious_swi (used in original swi_NI's last branch).
  map_NI: f_NI IRel IRel' f -> f_PU IRel IRel' f -> f_NI ORel ORel' g ->
          NI IRel' ORel p -> NI IRel ORel' (map f g p).
  f_NI ARel BRel h = forall l i i', rel ARel l i i' -> rel BRel l (h i)(h i').
  f_PU ARel BRel h = forall l i,    dis ARel l i    -> dis BRel l (h i).

1. eqpair_dom (V I) VRel IRel P Hclo : myrel [Times V I]
     Hclo : forall l x y, rel VRel l x.1 y.1 -> rel IRel l x.2 y.2 -> P x -> P y
     dis l x   := dis IRel l x.2 /\ P x
     rel l x y := (rel VRel l x.1 y.1 /\ rel IRel l x.2 y.2)
                  \/ ((dis IRel l x.2 /\ P x) /\ (dis IRel l y.2 /\ P y))
   *** INSIGHT (the crux): naive dis:=dis/\P breaks TRANSITIVITY. The fix is the
       closure Hclo: in a mixed rel-trans case (x rel y, y disclosed) you recover
       P x by applying Hclo to the SYMMETRIC rels (rel_sym) with P y. dis x.2 comes
       from dis_rel_dis2. This is exactly why the old eqpair_P used a vacuous
       witness instead. ***
   Arguments eqpair_dom {V I} VRel IRel P Hclo.  -- so all uses are positional.
   NOTE: rel/dis fields are INDEPENDENT of Hclo, so two eqpair_dom with different
   Hclo proofs are CONVERTIBLE for NI purposes (apply/eapply unify them fine).

2. Helpers:
   insert_oversize (A) a t n : size t < n -> insert n a t = t.   (elim t; destruct n)
   lthread_insert_rel : n <= size T -> lthread VRel f g l v (insert n (inl(w,i)) T)
        -> exists v', rel VRel l w (f i v').   (present-case extractor; elim n,
        generalize T v; head case gives the rel directly, recurse on tail state)
   projIl_size / projOl_size : size (projIl/projOl T) = size T.

3. Pf (V I) f vi := exists v', f (snd vi) v' = fst vi.   (kept; the sta-side predicate)

4. sta_conv'  -- same as library sta_conv but over (eqpair_dom VRel IRel (Pf f) Hclo);
   takes Hclo as an explicit param. Only diff from the eqpair_P version: input-case
   rel witness is "left. split;[apply Hw|apply rel_refl]" (was "left. left.").

5. sta_NI' : forall ... (Hclo : <Pf f closure>),
       fv_NI ORel VRel VRel g -> fv_NI IRel VRel VRel f -> f_EP IRel VRel f ->
       NI (eqpair_dom VRel IRel (Pf f) Hclo) ORel p ->
       NI IRel (eqpair VRel ORel) (sta f g v p).
   *** Hclo is a GENUINE separate side-condition the caller discharges; fv_NI f
       does NOT imply it. This is the real weakening over the original sta_NI. ***
   Clause 1 (rel): unchanged shape; rel left branch ("left; split;[rel_refl|Hii]").
   Clauses 2 & 3: CASE-SPLIT on (leqP n (size T)):
     - present (n<=size): get Pf f (w,i) via lthread_insert_rel + Hclo
       (Hclo l (f i v',i) (w,i) [rel_sym Hr] [rel_refl] [exists v']), then feed the
       eqpair_dom dis = (dis IRel l i /\ Pf f (w,i)) to p's clause 2/3.
     - absent (n>size): insert is a no-op (insert_oversize + projIl_size/projOl_size),
       so the inserted/removed element isn't really there — short-circuit straight to
       sta_conv' (clause 2) or to the already-equal Htp (clause 3). A bare
       "lthread ... -> Pf f (w,i)" helper would be FALSE here, which is why the
       split is necessary.
   VERIFIED: rocq_assumptions sta_NI' = standard_only (only Eq_rect_eq).

6. aware_false_closure {I} IRel {BRel} (Haware : forall l, aware BRel true l)
       : forall l (x y:[Times Bool I]), rel BRel l x.1 y.1 -> rel IRel l x.2 y.2
         -> x.1=false -> y.1=false.
   Proof: if y.1=true then rel BRel l false true, rel_sym, aware true gives
   true=false, discriminate. This IS the Hclo for eqpair_dom with P=(bi.1=false).

7. eqpair_dom_f_NI / eqpair_dom_f_PU  (the map-congruence keystone):
   given f_NI VRel BRel h and (forall x, P x -> Q (h x.1, x.2)), the value map
   (fun x => (h x.1, x.2)) is f_NI / f_PU from (eqpair_dom VRel IRel P) to
   (eqpair_dom BRel IRel Q). This is what lets map_NI carry eqpair_dom VRel I (Pf f)
   (produced by sta) to eqpair_dom BRel I (bool=false) (consumed by swi).
