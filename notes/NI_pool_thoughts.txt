Let's break down the problem and the solution step-by-step.

### 1. The Core Rule of Secrets in this Codebase
In this codebase, a security relation (`myrel`) has a rule called **Law 4**:
> If $A$ is secret, and $B$ is secret, then $A$ and $B$ **must be related**.

If we want the inputs to the process pool to be pairs of `(state, input)`:
* The **state** (which process is active) is public. So, two related pairs $(v_1, i_1)$ and $(v_2, i_2)$ must have the **same state** ($v_1 = v_2$).
* The **inputs** (from $I_1$ or $I_2$) are secret. So any pair $(v, \text{secret\_input})$ is a secret.

### 2. The Contradiction
Suppose we are at state `1` (process 1 is active) and receive a secret input $s_1$. The pair is `(1, s_1)`. This is a secret.
Later, we are at state `2` (process 2 is active) and receive a secret input $s_2$. The pair is `(2, s_2)`. This is also a secret.

By **Law 4**, since both `(1, s_1)` and `(2, s_2)` are secrets, they **must be related**.
But because the state is public, they **can only be related if their states are equal** (which would mean $1 = 2$, a contradiction).

This means you **cannot** write a standard `myrel` on `(state, input)` that allows secrets to happen at more than one active process index.

---

### 3. The Solution: Custom Property `NI_pool`
Instead of forcing the relation on `(state, input)` to be a standard `myrel` (which triggers this contradiction), we write a custom security property for the process pool $p$, called `NI_pool`. 

Instead of saying "the process $p$ is non-interfering under a relation", we describe the security behavior directly:
1. **Swapping:** If we swap two inputs on the **same** active process state $v$, the outputs remain related.
2. **Inserting/Removing:** If we insert or remove a secret input at **any** active process state $v$, the outputs remain related.

This avoids the contradiction because we never force `(1, s_1)` and `(2, s_2)` to be related to each other. We only require that inserting `(1, s_1)` at state `1` doesn't change the output, and inserting `(2, s_2)` at state `2` doesn't change the output.






Definition NI_pool_l (I O V : Ty) (IRel : myrel [I]) (VRel : myrel [V]) (ORel : myrel [O]) 
  (l : level) (p : Proc (Times V I) O) : Prop :=
  (forall (t : seq ([Times V I] + [O])) (v v' : [V]) (i i' : [I]) (n : nat),
     rel VRel l v v' ->
     rel IRel l i i' ->
     Trace (I := Times V I) (O := O) ORel l (insert n (inl (v, i)) t) p -> 
     Trace (I := Times V I) (O := O) ORel l (insert n (inl (v', i')) t) p) /\
  (forall (t : seq ([Times V I] + [O])) (v : [V]) (i : [I]) (n : nat),
     dis IRel l i -> 
     Trace (I := Times V I) (O := O) ORel l t p -> 
     Trace (I := Times V I) (O := O) ORel l (insert n (inl (v, i)) t) p) /\
  (forall (t : seq ([Times V I] + [O])) (v : [V]) (i : [I]) (n : nat),
     dis IRel l i -> 
     Trace (I := Times V I) (O := O) ORel l (insert n (inl (v, i)) t) p -> 
     Trace (I := Times V I) (O := O) ORel l t p).

Definition NI_pool (I O V : Ty) (IRel : myrel [I]) (VRel : myrel [V]) (ORel : myrel [O]) 
  (p : Proc (Times V I) O) : Prop :=
  forall l : level, NI_pool_l IRel VRel ORel l p.

Lemma NI_pool_reduceI : forall (I O V : Ty) (IRel : myrel [I]) (VRel : myrel [V]) (ORel : myrel [O])
  (p p' : Proc (Times V I) O) (vi : [Times V I]),
    NI_pool IRel VRel ORel p -> reduceI p vi p' -> NI_pool IRel VRel ORel p'.
Proof.
  intros I O V IRel VRel ORel p p' vi Hn Hred.
  move=> l. move: (Hn l). rewrite /NI_pool_l. ssa.
  - intros t v v' i i' n Hv Hi Htr.
    have Htr_p: Trace (I := Times V I) (O := O) ORel l (inl vi :: insert n (inl (v, i)) t) p.
    { eapply TR1; eauto. }
    have Heq: inl vi :: insert n (inl (v, i)) t = insert (n.+1) (inl (v, i)) (inl vi :: t) by [].
    rewrite Heq in Htr_p.
    have Htr_p_swap := H (inl vi :: t) v v' i i' (n.+1) Hv Hi Htr_p.
    have Heq2: inl vi :: insert n (inl (v', i')) t = insert (n.+1) (inl (v', i')) (inl vi :: t) by [].
    rewrite -Heq2 in Htr_p_swap.
    inv Htr_p_swap. match_dd.
    have <-: p'0 = p' by eapply reduceI_det; eauto.
    apply H6.
  - intros t v i n Hd Htr.
    have Htr_p: Trace (I := Times V I) (O := O) ORel l (inl vi :: t) p.
    { eapply TR1; eauto. }
    have Htr_p_ins := H0 (inl vi :: t) v i (n.+1) Hd Htr_p.
    have Heq: insert (n.+1) (inl (v, i)) (inl vi :: t) = inl vi :: insert n (inl (v, i)) t by [].
    rewrite Heq in Htr_p_ins.
    inv Htr_p_ins. match_dd.
    have <-: p'0 = p' by eapply reduceI_det; eauto.
    apply H6.
  - intros t v i n Hd Htr.
    have Htr_p: Trace (I := Times V I) (O := O) ORel l (inl vi :: insert n (inl (v, i)) t) p.
    { eapply TR1; eauto. }
    have Heq: inl vi :: insert n (inl (v, i)) t = insert (n.+1) (inl (v, i)) (inl vi :: t) by [].
    rewrite Heq in Htr_p.
    have Htr_p_rem := H1 (inl vi :: t) v i (n.+1) Hd Htr_p.
    inv Htr_p_rem. match_dd.
    have <-: p'0 = p' by eapply reduceI_det; eauto.
    apply H6.
Qed.

Lemma NI_pool_reduceO : forall (I O V : Ty) (IRel : myrel [I]) (VRel : myrel [V]) (ORel : myrel [O])
  (p p' : Proc (Times V I) O) (o : [O]),
    NI_pool IRel VRel ORel p -> reduceO p o p' -> NI_pool IRel VRel ORel p'.
Proof.
  intros I O V IRel VRel ORel p p' o Hn Hred.
  move=> l. move: (Hn l). rewrite /NI_pool_l. ssa.
  - intros t v v' i i' n Hv Hi Htr.
    have Hrefl: rel ORel l o o.
    { have: Equivalence (rel ORel l) := equiv ORel l. move=> Hequiv. reflexivity. }
    have Htr_p: Trace (I := Times V I) (O := O) ORel l (inr o :: insert n (inl (v, i)) t) p.
    { eapply TR2; eauto. }
    have Heq: inr o :: insert n (inl (v, i)) t = insert (n.+1) (inl (v, i)) (inr o :: t) by [].
    rewrite Heq in Htr_p.
    have Htr_p_swap := H (inr o :: t) v v' i i' (n.+1) Hv Hi Htr_p.
    have Heq2: inr o :: insert n (inl (v', i')) t = insert (n.+1) (inl (v', i')) (inr o :: t) by [].
    rewrite -Heq2 in Htr_p_swap.
    inv Htr_p_swap. match_dd.
    have [? <-] : o' = o /\ p'0 = p' by eapply reduceO_det; eauto.
    apply H7.
  - intros t v i n Hd Htr.
    have Hrefl: rel ORel l o o.
    { have: Equivalence (rel ORel l) := equiv ORel l. move=> Hequiv. reflexivity. }
    have Htr_p: Trace (I := Times V I) (O := O) ORel l (inr o :: t) p.
    { eapply TR2; eauto. }
    have Htr_p_ins := H0 (inr o :: t) v i (n.+1) Hd Htr_p.
    have Heq: insert (n.+1) (inl (v, i)) (inr o :: t) = inr o :: insert n (inl (v, i)) t by [].
    rewrite Heq in Htr_p_ins.
    inv Htr_p_ins. match_dd.
    have [? <-] : o' = o /\ p'0 = p' by eapply reduceO_det; eauto.
    apply H7.
  - intros t v i n Hd Htr.
    have Hrefl: rel ORel l o o.
    { have: Equivalence (rel ORel l) := equiv ORel l. move=> Hequiv. reflexivity. }
    have Htr_p: Trace (I := Times V I) (O := O) ORel l (inr o :: insert n (inl (v, i)) t) p.
    { eapply TR2; eauto. }
    have Heq: inr o :: insert n (inl (v, i)) t = insert (n.+1) (inl (v, i)) (inr o :: t) by [].
    rewrite Heq in Htr_p.
    have Htr_p_rem := H1 (inr o :: t) v i (n.+1) Hd Htr_p.
    inv Htr_p_rem. match_dd.
    have [? <-] : o' = o /\ p'0 = p' by eapply reduceO_det; eauto.
    apply H7.
Qed.

Lemma sta_conv_pool : forall (I O V : Ty) (VRel : myrel [V]) (IRel : myrel [I]) (ORel : myrel [O])
  (f : [I] -> [V] -> [V]) (g : [O] -> [V] -> [V]) (l : level) (L : seq ([Times V I] + [Times V O]))
  (p : Proc (Times V I) O) (v : [V]),
    fv_NI ORel VRel VRel g ->
    fv_NI IRel VRel VRel f ->
    NI_pool IRel VRel ORel p ->
    Trace ORel l (projOl L) p ->
    lthread VRel f g l v L ->
    Trace (eqpair VRel ORel) l (projIl L) (sta f g v p).
Proof.
  intros I O V VRel IRel ORel f g l L.
  elim: L => [|a L' IH] p v Hg Hf Hn Htr Hlt.
  - simpl. simpl in Htr. apply TR0.
  - destruct a as [wi|wo].
    + destruct wi as [v_in i].
      simpl in Htr.
      simpl in Hlt. ssa.
      have Htr_swap: Trace (I := Times V I) (O := O) ORel l (inl (f i v, i) :: projOl L') p.
      { move: Htr.
        have ->: inl (v_in, i) :: projOl L' = insert 0 (inl (v_in, i)) (projOl L') by [].
        have ->: inl (f i v, i) :: projOl L' = insert 0 (inl (f i v, i)) (projOl L') by [].
        move: (Hn l) => [Hswap _].
        apply: Hswap.
        - apply H.
        - have: Equivalence (rel IRel l) := equiv IRel l. move=> Hequiv. reflexivity.
      }
      simpl in Htr_swap.
      inv Htr_swap. match_dd.
      apply TR1 with (p' := sta f g (f i v) p') (i := i).
      * eapply reduce_staI; eauto.
      * eapply IH; eauto.
        -- eapply NI_pool_reduceI; eauto.
    + destruct wo as [v_out o].
      simpl in Htr.
      simpl in Hlt. ssa.
      inv Htr. match_dd.
      apply TR2 with (p' := sta f g (g o' v) p') (o' := (g o' v, o')).
      * eapply reduce_staO; eauto.
      * simpl. split.
        -- have Hequiv: Equivalence (rel VRel l) := equiv VRel l.
           eapply Equivalence_Transitive.
           ++ apply Hg with (i := o') (i' := o) (v := v) (v' := v); [apply H4 | reflexivity].
           ++ symmetry. apply H.
        -- apply H4.
      * eapply IH; eauto.
        -- eapply NI_pool_reduceO; eauto.
        -- eapply lthread_stable with (v := g o v); eauto.
Qed.

Lemma sta_NI_pool : forall (I O V : Ty) (p : Proc (Times V I) O) f g v 
  (IRel : myrel [I]) (VRel : myrel [V]) (ORel : myrel [O]),
    fv_NI ORel VRel VRel g -> fv_NI IRel VRel VRel f -> f_EP IRel VRel f ->
    NI_pool IRel VRel ORel p ->
    NI IRel (eqpair VRel ORel) (sta f g v p).
Proof.
  intros I O V p f g v IRel VRel ORel Hg Hf Hep Hn.
  rewrite /NI /NI_l.
  ssa.
  - intros t i i' n Hr Htr.
    apply (sta_proj_lthread Hf Hg) in Htr.
    destruct Htr as [T [Heq [Htr_p Hlt]]].
    symmetry in Heq.
    apply (projIl_insert_inv v) in Heq.
    destruct Heq as [w [T'' [Hdec Hproj]]].
    subst T.
    rewrite projOl_insert in Htr_p.
    have Htr_p_swap : Trace (I := Times V I) (O := O) ORel l (insert n (inl (w, i')) (projOl T'')) p.
    { move: (Hn l) => [Hswap _].
      apply Hswap with (v := w) (v' := w) (i := i) (i' := i') (n := n) (t := projOl T'').
      - have: Equivalence (rel VRel l) := equiv VRel l. move=> Hequiv. reflexivity.
      - exact Hr.
      - exact Htr_p. }
    have Heqp: inl (w, i') = projO (inl (w, i')) by [].
    rewrite Heqp in Htr_p_swap.
    rewrite -projOl_insert in Htr_p_swap.
    have Hlt_swap : lthread VRel f g l v (insert n (inl (w, i')) T'').
    { eapply lthread_swap; eauto. }
    have Htr_conv := sta_conv_pool Hg Hf Hn Htr_p_swap Hlt_swap.
    rewrite projIl_insert Hproj in Htr_conv.
    apply Htr_conv.
  - intros t i n Hd Htr.
    apply (sta_proj_lthread Hf Hg) in Htr.
    destruct Htr as [T [Heq [Htr_p Hlt]]].
    subst t.
    have Hlt_ins := lthread_insert_dis n Hf Hg Hep Hd Hlt.
    destruct Hlt_ins as [w Hlt_ins].
    have Htr_p_ins : Trace (I := Times V I) (O := O) ORel l (insert n (inl (w, i)) (projOl T)) p.
    { move: (Hn l) => [_ [Hins _]]. apply Hins; eauto. }
    have Heqp: inl (w, i) = projO (inl (w, i)) by [].
    rewrite Heqp in Htr_p_ins.
    rewrite -projOl_insert in Htr_p_ins.
    have Htr_conv := sta_conv_pool Hg Hf Hn Htr_p_ins Hlt_ins.
    rewrite projIl_insert in Htr_conv.
    have Heqi: inl i = projI (inl (w, i)) by [].
    rewrite Heqi.
    apply Htr_conv.
  - intros t i n Hd Htr.
    apply (sta_proj_lthread Hf Hg) in Htr.
    destruct Htr as [T [Heq [Htr_p Hlt]]].
    symmetry in Heq.
    apply (projIl_insert_inv v) in Heq.
    destruct Heq as [w [T'' [Hdec Hproj]]].
    subst T.
    rewrite projOl_insert in Htr_p.
    have Hlt_rem : lthread VRel f g l v T''.
    { eapply lthread_remove_dis; eauto. }
    have Heqp: inl (w, i) = projO (inl (w, i)) by [].
    rewrite Heqp in Htr_p.
    have Htr_p_rem : Trace ORel l (projOl T'') p.
    { move: (Hn l) => [_ [_ Hrem]]. apply Hrem with (v := w) (i := i) (n := n); eauto. }
    have Htr_conv := sta_conv_pool Hg Hf Hn Htr_p_rem Hlt_rem.
    rewrite Hproj in Htr_conv.
    apply Htr_conv.
Qed.
